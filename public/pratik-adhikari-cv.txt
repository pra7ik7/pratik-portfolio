%========================= RESEARCH EXPERIENCE =======================
\section{Research Experience}

\entry{Digital Twin for Real-Time Monitoring and Short-Term Forecasting
of a Grid-Connected PV System}{Sep 2025 -- Present}
\role{Final-year project (four-member team), Department of Electrical
Engineering, Pulchowk Campus. Supervisors: Dr.\ Basanta Kumar Gautam (Pulchowk Campus) and
Er.\ Deependra Neupane (Electricity Regulatory Commission).}
\begin{itemize}
  \item Built a converter-level MATLAB/Simulink twin of a 335~W
  grid-connected PV module (single-diode model, DC--DC stage with
  perturb-and-observe MPPT, single-phase inverter) and validated it
  against 41 field measurements on a clear and a partly cloudy day.
  \item Verified MPPT operation against the module's own P--V curve
  (99.9--100\% tracking efficiency, 200--1000~W/m$^2$). Against the 41
  field readings the twin gives 24.2~W RMSE ($r = 0.969$) with on-site
  irradiance and 99.8~W ($r = 0.69$) with satellite-derived irradiance,
  showing the value of site-specific sensing. On the clear day, measured
  power was 87.6\% of the twin's prediction, indicative of unmodelled
  losses.
  \item Designed and calibrated a low-cost Arduino/LDR irradiance sensor
  against a smartphone-based reference (power-law fit, $R^2 = 0.974$,
  $n = 21$) and used it to drive the twin from live serial data in a
  real-time demonstration.
  \item Developed a SARIMA$(1,1,1)\times(1,1,1)_{24}$ day-ahead forecaster
  for irradiance and temperature and propagated it through the twin: on a
  held-out day, daily energy was within 0.8\% of that obtained with
  measured weather, against 6.7\% for a persistence baseline.
\end{itemize}

\vspace{3pt}
\entry{Deadbeat Predictive Current Control and Passive Damping of an
LCL-Filtered Grid-Connected Converter}{May 2026 -- Present}
\role{Simulation study with Panas Bhattarai (Dept.\ of Electrical
Engineering, Tribhuvan University), who provided guidance; I lead the
modeling, simulation, analysis and writing.}
\begin{itemize}
  \item Developed a stationary $\alpha\beta$-frame deadbeat predictive
  current controller for a 4.1~kW, 380~V two-level voltage-source
  converter with an LCL filter (2.478~kHz resonance): power commands map
  algebraically to current references, needing no Park transformation or
  synchronous-frame regulator. Simulated with 10~kHz SVPWM and one control
  period of computation delay.
  \item Determined how much passive damping the controller needs: the
  undamped converter does not operate (voltage limiter active at every
  step, power flow reverses), no deadbeat gain from 0.05 to 1 restores
  operation, and the minimum damping ratio lies between 0.034 and 0.057.
  \item Identified a damping range of 21--30~$\Omega$ per delta branch
  that operates under $\pm$20\% $L_1$, $L_2$, $C$ tolerance and up to
  5~mH added grid inductance, which the 10~$\Omega$ minimum does not.
  Worst harmonic-limit utilisation is 0.34--0.23 (IEC~61000-4-7 subgroups
  against IEEE~519-2022) at 0.15--0.21\% dissipation, about 14 times the
  fundamental-frequency estimate.
  \item Earlier study (5~kW, 400~V, no computation delay): 60~$\Omega$
  per branch removed 91.8\% of the resonance-band current; subgroup TDD
  3.0--4.9\% across six operating points (at most 5.1\% over ten
  robustness cases). Preceded by FCS-MPC and L versus LCL comparisons.
\end{itemize}
